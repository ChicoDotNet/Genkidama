cmake_minimum_required(VERSION 3.25)
project(TelemetryTape VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 23)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_library(telemetry src/telemetry.c)
target_include_directories(telemetry PUBLIC include)
add_executable(telemetry_cli src/main.c)
target_link_libraries(telemetry_cli PRIVATE telemetry)
enable_testing()
add_executable(telemetry_tests tests/test_telemetry.c)
target_link_libraries(telemetry_tests PRIVATE telemetry)
add_test(NAME telemetry_tests COMMAND telemetry_tests)
foreach(target telemetry telemetry_cli telemetry_tests)
  if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endforeach()
include(GNUInstallDirs)
install(TARGETS telemetry telemetry_cli
  ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}
  RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES include/telemetry.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
